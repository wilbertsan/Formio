```ts
import { z } from 'zod'

export const modelParametersSchema = z.object({
  width: z.number().min(10).max(100),
  height: z.number().min(10).max(100),
  depth: z.number().min(10).max(100),
  radius: z.number().min(0).max(10),
  segments: z.number().int().min(2).max(32),
})

export type ModelParameters = z.infer<typeof modelParametersSchema>

export type ParameterKey = keyof ModelParameters

export interface ParameterDefinition {
  key: ParameterKey
  label: string
  min: number
  max: number
  step: number
  unit: 'mm'
}

export const parameterDefinitions: ParameterDefinition[] = [
  { key: 'width', label: 'Largura', min: 10, max: 100, step: 1, unit: 'mm' },
  { key: 'height', label: 'Altura', min: 10, max: 100, step: 1, unit: 'mm' },
  { key: 'depth', label: 'Profundidade', min: 10, max: 100, step: 1, unit: 'mm' },
  { key: 'radius', label: 'Raio de arredondamento', min: 0, max: 10, step: 1, unit: 'mm' },
  { key: 'segments', label: 'Segmentos', min: 2, max: 32, step: 1, unit: 'mm' },
]

export const defaultParameters: ModelParameters = {
  width: 60,
  height: 40,
  depth: 30,
  radius: 4,
  segments: 8,
}

export function clampParameter(key: ParameterKey, value: number): number {
  const def = parameterDefinitions.find((d) => d.key === key)
  if (!def) return value
  const stepped = Math.round(value / def.step) * def.step
  return Math.min(def.max, Math.max(def.min, stepped))
}
```

```ts
import { useCallback, useReducer } from 'react'
import {
  clampParameter,
  defaultParameters,
  type ModelParameters,
  type ParameterKey,
} from '../config/parameters'

type ModelParametersAction =
  | { type: 'update'; key: ParameterKey; value: number }
  | { type: 'reset' }

function modelParametersReducer(
  state: ModelParameters,
  action: ModelParametersAction,
): ModelParameters {
  switch (action.type) {
    case 'update':
      return { ...state, [action.key]: clampParameter(action.key, action.value) }
    case 'reset':
      return defaultParameters
  }
}

export function useModelParameters() {
  const [parameters, dispatch] = useReducer(modelParametersReducer, defaultParameters)

  const updateParameter = useCallback((key: ParameterKey, value: number) => {
    dispatch({ type: 'update', key, value })
  }, [])

  const resetParameters = useCallback(() => {
    dispatch({ type: 'reset' })
  }, [])

  return { parameters, updateParameter, resetParameters }
}
```

```tsx
import { View } from '@aws-amplify/ui-react'

export type StudioTool = 'select' | 'move' | 'rotate' | 'zoom'

interface StudioToolbarProps {
  activeTool: StudioTool
  onSelectTool: (tool: StudioTool) => void
  onResetCamera: () => void
}

const tools: { id: StudioTool; label: string; icon: string }[] = [
  { id: 'select', label: 'Selecionar', icon: '⬚' },
  { id: 'move', label: 'Mover', icon: '✥' },
  { id: 'rotate', label: 'Rotacionar', icon: '⟳' },
  { id: 'zoom', label: 'Zoom', icon: '⤢' },
]

export function StudioToolbar({ activeTool, onSelectTool, onResetCamera }: StudioToolbarProps) {
  return (
    <View
      as="nav"
      aria-label="Ferramentas do studio"
      className="flex w-14 flex-col items-center gap-1 border-r border-neutral-800 bg-neutral-900 py-2"
    >
      {tools.map((tool) => (
        <button
          key={tool.id}
          type="button"
          title={tool.label}
          aria-label={tool.label}
          aria-pressed={activeTool === tool.id}
          onClick={() => onSelectTool(tool.id)}
          className={`flex h-10 w-10 items-center justify-center rounded-md text-lg transition-colors ${
            activeTool === tool.id
              ? 'bg-teal-600 text-white'
              : 'text-neutral-400 hover:bg-neutral-800 hover:text-neutral-100'
          }`}
        >
          <span aria-hidden="true">{tool.icon}</span>
        </button>
      ))}
      <View className="my-1 h-px w-8 bg-neutral-800" />
      <button
        type="button"
        title="Resetar câmera"
        aria-label="Resetar câmera"
        onClick={onResetCamera}
        className="flex h-10 w-10 items-center justify-center rounded-md text-lg text-neutral-400 transition-colors hover:bg-neutral-800 hover:text-neutral-100"
      >
        <span aria-hidden="true">⌂</span>
      </button>
    </View>
  )
}
```

```tsx
import { Suspense, useEffect, useRef } from 'react'
import type { OrbitControls as OrbitControlsImpl } from 'three-stdlib'
import { Canvas } from '@react-three/fiber'
import { Environment, Grid, OrbitControls } from '@react-three/drei'
import { SampleSolid } from './SampleSolid'
import type { ModelParameters } from '../config/parameters'

interface StudioViewportProps {
  parameters: ModelParameters
  resetCameraSignal: number
}

export function StudioViewport({ parameters, resetCameraSignal }: StudioViewportProps) {
  const controlsRef = useRef<OrbitControlsImpl>(null)

  useEffect(() => {
    if (resetCameraSignal > 0) {
      controlsRef.current?.reset()
    }
  }, [resetCameraSignal])

  return (
    <div
      role="img"
      aria-label="Visualização 3D do modelo"
      className="relative flex-1 bg-neutral-950"
    >
      <Canvas
        shadows
        camera={{ position: [90, 70, 90], fov: 45 }}
        className="absolute inset-0"
      >
        <Suspense fallback={null}>
          <ambientLight intensity={0.5} />
          <directionalLight
            position={[80, 120, 60]}
            intensity={1.5}
            castShadow
            shadow-mapSize={[1024, 1024]}
          />
          <SampleSolid parameters={parameters} />
          <Grid
            args={[300, 300]}
            cellSize={10}
            cellColor="#404040"
            sectionSize={50}
            sectionColor="#525252"
            fadeDistance={280}
            infiniteGrid
          />
          <Environment preset="city" />
          <OrbitControls ref={controlsRef} makeDefault />
        </Suspense>
      </Canvas>
    </div>
  )
}
```

Nota: o `useEffect` com `resetCameraSignal` chama `controlsRef.current?.reset()` — no teste o OrbitControls é mockado, então o efeito é um no-op seguro. `three-stdlib` é dependência transitiva do drei (o tipo `OrbitControls` vem de lá); se o typecheck reclamar, troque o tipo do ref para `React.ComponentRef<typeof OrbitControls>`.
